// Closes the Electron app listening on CDP_PORT with Browser.close on the browser target.
const port = Number(process.env.CDP_PORT ?? 9344);
const v = (await (await fetch(`http://127.0.0.1:${port}/json/version`)).json()) as { webSocketDebuggerUrl: string };
const ws = new WebSocket(v.webSocketDebuggerUrl);
await new Promise((r) => ws.addEventListener("open", r, { once: true }));
ws.addEventListener("message", (e) => console.log("reply", String(e.data)));
ws.addEventListener("close", () => console.log("socket closed"));
ws.send(JSON.stringify({ id: 1, method: "Browser.close" }));
await new Promise((r) => setTimeout(r, 3000));
process.exit(0);
