const records: { part: string; what: string; block: string; value: unknown }[] = [];
const json = (value: unknown) =>
  JSON.stringify(value, (_, v) => (typeof v === "bigint" ? v.toString() : v));
let B = 0n;
function rec(part: string, what: string, value: unknown, block: bigint | string = B) {
  records.push({ part, what, block: String(block), value });
  process.stdout.write(`[${part}] @${block} ${what}: ${json(value)}\n`);
}
async function attempt<T>(run: () => Promise<T>): Promise<T | { error: string }> {
  try {
    return await run();
  } catch (cause) {
    return { error: String(cause instanceof Error ? cause.message : cause).split("\n")[0] ?? "" };
  }
}
const read = (address: Address, abi: Abi, functionName: string, args: unknown[] = []) =>
  attempt(() =>
    client.readContract({ address, abi, functionName, args, blockNumber: B } as never),
  ) as Promise<unknown>;
const codeSize = async (address: Address) => {
  const code = await client.getCode({ address, blockNumber: B });
  return code === undefined || code === "0x" ? 0 : (code.length - 2) / 2;
};

// Royalty accounting: every Royalty / Claimed the hook emitted, and each swap it came from.
async function royalties(label: string, hook: Address, from: bigint, pools: { name: string; poolId: Hex; key: PoolKey }[]) {
  const [roy, claimed] = await Promise.all([
    client.getLogs({ address: hook, event: royaltyEvent, fromBlock: from, toBlock: B }),
    client.getLogs({ address: hook, event: claimedEvent, fromBlock: from, toBlock: B }),
  ]);
  rec("3-market", `${label} Royalty events`, roy.map((l) => ({ world: l.args.world, currency: l.args.currency, amount: formatEther(l.args.amount ?? 0n), block: l.blockNumber, tx: l.transactionHash })), `${from}..${B}`);
  rec("3-market", `${label} Claimed events`, claimed.map((l) => ({ world: l.args.world, currency: l.args.currency, to: l.args.to, amount: formatEther(l.args.amount ?? 0n), block: l.blockNumber, tx: l.transactionHash })), `${from}..${B}`);
  for (const pool of pools) {
    const swaps = await client.getLogs({
      address: UNISWAP_SEPOLIA.poolManager,
      event: swapEvent,
      args: { id: pool.poolId },
      fromBlock: from,
      toBlock: B,
    });
    for (const s of swaps) {
      const inTx = roy.filter((r) => r.transactionHash === s.transactionHash);
      const amounts = [s.args.amount0 ?? 0n, s.args.amount1 ?? 0n];
      // Exact input: the positive delta (to the swapper) is the output side the hook takes 1% of.
      const out = amounts.find((a) => a > 0n) ?? 0n;
      const royaltyTotal = inTx.reduce((sum, r) => sum + (r.args.amount ?? 0n), 0n);
      rec("3-market", `${label} swap in ${pool.name}`, {
        tx: s.transactionHash,
        block: s.blockNumber,
        amount0: s.args.amount0,
        amount1: s.args.amount1,
        poolOutput: formatEther(out),
        royalties: inTx.map((r) => ({ world: r.args.world, amount: formatEther(r.args.amount ?? 0n) })),
        royaltyTotal: formatEther(royaltyTotal),
        royaltyPerMillionOfOutput: out === 0n ? null : Number((royaltyTotal * 1_000_000n) / out),
        buyerGot: formatEther(out - royaltyTotal),
      }, s.blockNumber);
    }
  }
}

rec("3-market", "v1 aether-land worldOf / poolKeyOf", { world: v1World, key: v1Key });
if (v1Key && "currency0" in v1Key) {
  await royalties("v1", V1.hook, V1.from, [{ name: "aether-land (v1)", poolId: poolId(v1Key), key: v1Key }]);
}

// ── 4. Gas station and operator balances ────────────────────────────────────────────────────
const gasPrice = await client.getGasPrice();
rec("4-station", "eth_gasPrice (gwei)", formatUnits(gasPrice, 9), "latest");
for (const [what, address] of [["station key", STATION], ["operator key", OPERATOR]] as const) {
  const [balance, nonce] = await Promise.all([
    client.getBalance({ address, blockNumber: B }),
    client.getTransactionCount({ address, blockNumber: B }),
  ]);
  rec("4-station", `${what} ${address}`, { balanceEth: formatEther(balance), nonce });
}

writeFileSync(process.argv[2] ?? "audit.json", json({ block: B, records }));
process.stdout.write(`\n(read-only; pinned block ${B}; nothing was sent)\n`);
